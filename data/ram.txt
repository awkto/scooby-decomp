# RAM / I/O map: addr size name   # notes.  Emitted as .equ in src/scooby.s; absolute operands that
# fall inside a named range are written symbolically (ram_x or ram_x+off).  Names must be unique
# assembler identifiers.  Sources: docs/formats.md, docs/sound.md, code reading ("?" = guess).
# Sizes are the access width / extent; a multi-byte entry also names the bytes inside it as name+N.

# ---- Genesis hardware
A00000 8192 z80_ram          # Z80 RAM window (68k bus)
A00036 1    z80_ring_wr      # sound mailbox write index (owned by the 68k)
A01B22 1    z80_vars         # driver variables ($1B22+n, see sound_get_var)
A01B40 64   z80_ring         # sound mailbox ring buffer
A10001 1    io_version       # version register
A10003 1    io_data1         # controller port 1 data
A10005 1    io_data2         # controller port 2 data
A10008 4    io_ctrl_both     # TMSS-style check at reset reads ctrl1/ctrl2 as a long
A10009 1    io_ctrl1
A1000B 1    io_ctrl2
A1000C 2    io_ctrl3
A11100 2    z80_busreq
A11200 2    z80_resetreg
C00000 4    vdp_data
C00004 4    vdp_ctrl
C00008 2    vdp_hv           # HV counter

FF0008 640  ram_sat_buf      # sprite attribute table, DMA'd to VRAM $D800 every frame
FF000C 2    ram_8828_index   # byte offset into dat_008828 (used by the big tile blitters)

FF0428 4    ram_rng_state    # seeded 45C3F2D1 at game_init, stepped by rng_next
FF0484 12   ram_list_484     # six-entry list walked at $00B0F8 (a6) ?

# ---- input / flags
FF09DC 1    ram_vblank_flags # bit0 = wait_vblank handshake (set by main loop, cleared by VBlank); other bits = deferred VBlank jobs
FF09DE 2    ram_09de
FF09E0 1    ram_pad1         # pad 1 raw, ACTIVE LOW: bit7 S, 6 A, 5 C, 4 B, 3 R, 2 L, 1 D, 0 U (0 = pressed)

FF09E2 2    ram_pad_last     # copy of ram_pad1/2 taken at the start of the previous pad_read
FF09E4 2    ram_pad_ref      # reference copy for change detection (written elsewhere)
FF09E6 1    ram_pad1_edge    # pad 1 bits that changed vs ram_pad_ref, OR-accumulated by pad_read (consumer clears)
FF09E7 1    ram_pad2_edge
FF09E8 1    ram_09e8
FF09E9 1    ram_09e9
FF09EA 1    ram_09ea
FF09EB 1    ram_09eb
